### Task 6: Create DocsPage.tsx

**Files:**
- Create: `web/src/pages/DocsPage.tsx`

- [ ] **Step 1: Create DocsPage.tsx**

Uses `import.meta.glob` to dynamically load MDX components. Falls back to a "not found" message if the page doesn't exist.

```tsx
import { useParams, useNavigate } from 'react-router-dom';
import { useEffect } from 'react';
import ReactMarkdown from 'react-markdown';
import remarkGfm from 'remark-gfm';
import { Prism as SyntaxHighlighter } from 'react-syntax-highlighter';
import { oneDark } from 'react-syntax-highlighter/dist/esm/styles/prism';

const modules = import.meta.glob('../../docs/**/*.mdx');

export default function DocsPage() {
  const { section, slug } = useParams<{ section: string; slug: string }>();
  const navigate = useNavigate();

  if (!section || !slug) {
    navigate('/docs/user/getting-started', { replace: true });
    return null;
  }

  const key = `../../docs/${section}/${slug}.mdx`;
  const loader = modules[key];

  if (!loader) {
    return (
      <div className="text-center py-20">
        <h1 className="text-2xl font-bold mb-4">404</h1>
        <p className="text-base-content/60">Document not found</p>
        <button onClick={() => navigate('/docs/user/getting-started')} className="btn btn-primary mt-4">
          Go to Home
        </button>
      </div>
    );
  }

  // Lazy load the MDX module
  const [Content, setContent] = React.useState<React.ComponentType | null>(null);

  useEffect(() => {
    loader().then((mod: any) => setContent(() => mod.default));
  }, [loader]);

  if (!Content) {
    return (
      <div className="flex items-center justify-center py-20">
        <span className="loading loading-spinner loading-lg" />
      </div>
    );
  }

  return <Content />;
}
```

**Note:** The `import.meta.glob` path is relative to this file's location at `web/src/pages/DocsPage.tsx`. Going up two levels (`../../`) reaches `web/`, so `../../docs/**/*.mdx` maps to `web/docs/**/*.mdx`.

# 快速开始

欢迎使用 LLM Gateway！本指南将帮助你快速上手。

## 创建 API 密钥

1. 登录控制台，进入 **API 密钥** 页面
2. 点击「创建密钥」按钮
3. 设置密钥名称，点击确认
4. 复制生成的 API 密钥（创建后无法再次查看）

## 基本使用

获取 API 密钥后，你可以通过 OpenAI 兼容的 SDK 接入：

```js
import OpenAI from 'openai';

const client = new OpenAI({
  apiKey: 'sk-your-api-key',
  baseURL: 'https://your-gateway-url/v1',
});

const chat = await client.chat.completions.create({
  model: 'gpt-4o',
  messages: [{ role: 'user', content: 'Hello!' }],
});
```

## Anthropic 协议

也支持 Anthropic 协议：

```js
import Anthropic from '@anthropic-ai/sdk';

const client = new Anthropic({
  apiKey: 'sk-your-api-key',
  baseURL: 'https://your-gateway-url/v1',
});

const message = await client.messages.create({
  model: 'claude-3-5-sonnet-20241022',
  max_tokens: 100,
  messages: [{ role: 'user', content: 'Hello!' }],
});
```
```
